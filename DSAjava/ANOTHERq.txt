/*
Approach
Time Complexity: O(n)
Space Complexity: O(1)
1.Created an array to make count of alphabets
2. took numerical equivalent of all alphabets and incremented the occurences with one array, decremented the occurences with the other.
3. Checked for net number of occurences. If all are zero, it is an anagram.

*/

class Solution {
    public boolean isAnagram(String s, String t) {
        int[] a= new int[26];
        if(s.length()!=t.length())
            return false;
        for(int i=0;i<s.length();i++){
            a[s.charAt(i)-97]++;
            a[t.charAt(i)-97]--;
        }
        for(int i=0;i<26;i++)
           if(a[i]!=0)
              return false;
        return true;
        
    }
    
    
}
